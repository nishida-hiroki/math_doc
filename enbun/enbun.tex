\section{円分体}
\begin{definition}
    $\zeta^m=1$を満たす元$\zeta$を
    $1$\textbf{の}$m$\textbf{乗根}という．
    $1$の$m$乗根$\zeta$で$1\leq d<m$なる
    整数$d$について$\zeta^d\neq 1$となるものを
    $1$\textbf{の原始}$m$\textbf{乗根}という．
\end{definition}

\begin{lemma}
    $\Omega$を代数的閉体とするとき次は同値である．
    \begin{enumerate}
        \item $1$の原始$m$乗根が存在する．
        \item $1$の$m$乗根全体が位数$m$の巡回群をなす．
        \item $\Omega$の標数を$p$とするとき$p$は$m$と互いに素である．または$p$は$0$である．
    \end{enumerate}
\end{lemma}

\begin{proof}
    $2\Rightarrow 1$：巡回群の生成元が$1$の原始$m$乗根である．
    $1\Rightarrow 3$：対偶を示す．$m=pn\, (n\in \mathbb{N})$と表すことができる．
    $\zeta$を$m$乗根とすると$\zeta^m=\zeta^{pn}=(\zeta^n)^p=1$である．
    $X^p-1=(X-1)^p$であることから$\zeta^n=1$となり，
    $1$の原始$m$乗根は存在しない．
    $3\Rightarrow 2$：
    $f(X)=X^m-1$の根を考えるとこれは$1$の$m$乗根である．微分$f'(X)=mX^{m-1}$は仮定より$0$でない．
    よって$f$は分離多項式であり，
    $f(X)$の根全体の集合を$G$とおくと
    $G$は位数$m$の群である．$d\mid m$となる整数$d$に対して，
    位数が$d$の$G$の部分群$H$の個数を考える．
    このような$H$が存在すると仮定して，$H$の任意の元$a$は
    フェルマーの小定理より$a^d=1$である．
    すなわち$g(X)=X^d-1$の根である．位数$d$の異なる部分群$H'$を
    考えてもやはり$g(X)$の根となることから$H$の取り方は
    高々一通りしかなく，すなわち$G$は位数$m$の巡回群である．
\end{proof}

\begin{theorem}
    $F$を任意の体，$\overline{F}$を$F$の代数的閉包とする．$\zeta_m\in 
    \overline{F}$を$1$の原始$m$乗根とすれば，$F(\zeta_m)/F$は
    アーベル拡大であり，$\mathrm{Gal}(F(\zeta_m)/F)\subset 
    (\mathbb{Z}/m\mathbb{Z})^\times$が成り立つ．
\end{theorem}

\begin{proof}
    $\zeta_m$は$X^m-1$が分離的であることから$F(\zeta_m)/F$は分離拡大
    である．$\zeta_m$の最小多項式の根全体が
    $\langle\zeta_m\rangle$であることから
    正規拡大であることもわかり，
    $F(\zeta_m)/F$はガロア拡大である．
    $\sigma \in \mathrm{Gal}(F(\zeta_m)/F)$を一つとる．
    $\sigma$は$X^m-1$の根をうつすから，ある整数
    $s\, (1\leq s \leq m-1)$が存在して
    $\sigma(\zeta_m)=\zeta_m^s$となる．
    また，$\tau \in \mathrm{Gal}(F(\zeta_m)/F),\, \tau(\zeta_m)=\zeta_m^t$
    に対して
    $\sigma\circ\tau (\zeta_m)=\sigma(\zeta_m^t)
    =\sigma(\zeta_m)^t=\zeta_m^{st}=\zeta_m^{ts}
    =\tau(\zeta_m)^s=\tau(\zeta_m^s)=\tau\circ\sigma (\zeta_m)$
    であるからアーベル拡大である．
    $\sigma \mapsto s$とする写像は単射である．実際核を考えれば
    $\sigma(\zeta_m)=\zeta_m^1\Rightarrow \sigma=\mathrm{id}_F$である．
    この対応が求める準同型である．
\end{proof}

\begin{definition}

    整数$1,2,\ldots,m$のなかで$m$と互いに素な整数の個数を$\phi (m)$と書き，
    これを\textbf{オイラーの関数}という．ただし$\phi (1)=1$とする．
\end{definition}

\begin{lemma}
    $\phi (m)= |(\mathbb{Z}/m\mathbb{Z})^\times|$
    であり，標数$p$が$m$と互いに素なら$\phi (m)$
    は$1$の原始$m$乗根の数に等しい．
\end{lemma}

\begin{proof}
    零環の可逆元の個数は$1$とみなす．前半は明らか．
    標数$p$と$m$が互いに素のとき$1$の原始$m$乗根が
    存在する．これを$\zeta_m$とする．$1$の$m$乗根
    $\{ \zeta_m,\zeta_m^2,\ldots, \zeta_m^m \}$
    のうち原始$m$乗根でないものの数を数える．
    $m$乗根であって原始$m$乗根でないものを
    $\zeta_m^s$とすると，ある整数$1\leq d < m$
    が存在して$\zeta_m^{sd}=1$となる．
    ここで$sd$は$m$の倍数である．
    仮にそうでない場合$t\equiv sd,\, (\text{mod}\, m)$を
    とれば$\zeta_m^t=1$となり原始であることと矛盾する．
    よって$s$は零因子であり，個数は一致する．
\end{proof}

\begin{definition}
    $1$の原始$m$乗根全体を根にする最高次係数$1$の多項式$\Phi_m(X)$を
    \textbf{円周等分多項式}という．また，$\Phi_m(X)=0$を
    \textbf{円周等分方程式}という．
\end{definition}

$\Omega$を代数閉体として$\sigma$を任意のその体自己同型写像とすると
$1$の原始$m$乗根$\zeta_m$に対して
$\sigma (\zeta_m)$も$1$の原始$m$乗根である．
そうでなければ$\sigma (\zeta_m)^d=\sigma(\zeta_m^d)=1\Rightarrow\zeta_m^d=1$
となって矛盾する．つまり$\sigma$は$1$の原始$m$乗根の置換である．
つまり円周等分多項式は$\sigma$で不変である．
さらにすべての自己同型写像の不変体は素体であるから
円周等分多項式の係数は素体の元である．
$1$の$m$乗根は$d\mid m$について$1$の$d$乗根でもあるから次が成り立つ．
\[ X^m-1=\Pi_{d\mid m}\Phi_d(X)\]
$d$の小さい順に$\Phi_d(X)$を考えるといい．
\begin{lemma}\label{lemma:ensyu toubun takousiki keisuu seisuu}
    素体が$\mathbb{Q}$の体の円周等分多項式の係数は整数である．
\end{lemma}

\begin{proof}
    $\Phi_m(X)$の$m$に関する帰納法で示す．
    $m=1$のときは明らかに原始多項式（ガウスの補題の方の）である．
    $m$が$2$以上のとき$d<m$なるすべての$d$で$\Phi_d(X)$が原始多項式
    であることを仮定する．ここで
    \[
    \Phi_m(X)=\frac{X^m-1}{\Pi_{d\mid m,d<m}\Phi_d(X)}
    \]
    を考える．帰納法の仮定とガウスの補題（原始多項式の積は原始多項式）
    より右辺の分母は原始多項式である．
    分子も明らかに原始多項式である．
    分母と分子がモニックであることから$\Phi_m(X)$もモニックである．
    $\Phi_m(X)$の各係数の分母の最小公倍数を$a$とおく．
    さらに$\displaystyle f(X)\coloneqq a\cdot\Phi_m(X)$
    とおくと$f(X)$は原始多項式である．これを用いると
    \[
    f(X)\cdot \Pi_{d\mid m,d<m}\Phi_d(X)=a(X^m-1)
    \]
    となる．左辺のどちらも原始多項式であるから，
    ガウスの補題より右辺も原始多項式である．
    この条件を満たす$a$は$1$のみである．
    よって$\Phi_m(X)$は整数係数である．
\end{proof}
上の証明は冗長である．
モニックな整数係数をモニックな整数係数で割りきれている
のだから商もモニックな整数係数となる．
そのように証明したほうが楽だが，ガウスの補題の練習のために
こうした．
\begin{theorem}
    体の標数を$0$とする．このとき$\Phi_m(X)$は$\mathbb{Q}$上既約である．
\end{theorem}
\begin{proof}
    補題\uref{lemma:ensyu toubun takousiki keisuu seisuu}より
    円周等分多項式は整数係数である．
\end{proof}



\section{計算}

$\mathbb{Q}$に$\zeta_7$を添加した体
$\mathbb{Q}(\zeta_7)$を考える．
$\zeta_7$の$\mathbb{Q}$上の最小多項式は
$X^6+X^5+X^4+X^3+X^2+X+1$であるから
$\mathbb{Q}(\zeta_7)/\mathbb{Q}$
は$6$次ガロア拡大である．
ガロアの基本定理により中間体$\mathbb{Q}\subset
K \subset \mathbb{Q}(\zeta_7)$で
$K/\mathbb{Q}$が$3$次巡回拡大となるものが存在する．
それを求めてみる．ガロア群の元としての同型写像は
$\sigma_i(\zeta_7)=\zeta_7^i,\,(i=1,\ldots,6)$
の$6$つである．以下その計算表である．


\subsection*{計算表}

\begin{tabular}{c|cccccc}
   & $\mathrm{id}$ & $\sigma_2$ & $\sigma_3$ & $\sigma_4$ & $\sigma_5$ & $\sigma_6$ \\ \hline
  $\mathrm{id}$ & $\mathrm{id}$ & $\sigma_2$ & $\sigma_3$ & $\sigma_4$ & $\sigma_5$ & $\sigma_6$ \\
  $\sigma_2$ &  & $\sigma_4$ & $\sigma_6$ & $\mathrm{id}$ & $\sigma_3$ & $\sigma_5$ \\
  $\sigma_3$ &  &  & $\sigma_2$ & $\sigma_5$ & $\mathrm{id}$ & $\sigma_4$ \\
  $\sigma_4$ &  &  &  & $\sigma_2$ & $\sigma_6$ & $\sigma_3$ \\
  $\sigma_5$ &  &  &  &  & $\sigma_4$ & $\sigma_2$ \\
  $\sigma_6$ &  &  &  &  &  & $\mathrm{id}$ \\
\end{tabular}


$\left< \sigma_6 \right>=\{\mathrm{id},\sigma_6\}\\
\left< \sigma_2 \right>=
\left< \sigma_4 \right>=\{\mathrm{id},\sigma_2,\sigma_4\}\\
\left< \sigma_3 \right>=
\left< \sigma_5 \right>=\{\mathrm{id},\sigma_2,\sigma_3,
\sigma_4,\sigma_5,\sigma_6\}$

これによって$\left< \sigma_6 \right>$
で固定される体が求める中間体であることがわかる．
$\displaystyle
\zeta_7+\zeta_7^6=2\mathrm{cos}\frac{2\pi}{7},\,
\zeta_7^2+\zeta_7^5=2\mathrm{cos}\frac{4\pi}{7},\,
\zeta_7^3+\zeta_7^4=2\mathrm{cos}\frac{6\pi}{7}
$は$\sigma_6$で不変であるから，これらを
追加した体$\displaystyle\mathbb{Q}(2\mathrm{cos}\frac{2\pi}{7})$
などが求めるものである．ガロアの基本定理により
中間体は一つであるから
\[
    \mathbb{Q}(2\mathrm{cos}\frac{2\pi}{7})=
    \mathbb{Q}(2\mathrm{cos}\frac{4\pi}{7})=
    \mathbb{Q}(2\mathrm{cos}\frac{6\pi}{7})
\]
である．
$\displaystyle 2\mathrm{cos}\frac{2\pi}{7}$
の$\mathbb{Q}$上の最小多項式も求めてみる．
$x=\zeta_7+\zeta_7^{-1}$とおく．
\[
\zeta_7^6+\zeta_7^5+\zeta_7^4+\zeta_7^3+\zeta_7^2+\zeta_7+1=0
\]
の両辺に$\zeta_7^{-3}$を掛けると
\[
\zeta_7^3+\zeta_7^2+\zeta_7^1+1+\zeta_7^{-1}+\zeta_7^{-2}+\zeta_7^{-3}=0
\]
になり，これに$x^2-2=\zeta_7^2+\zeta_7^{-2},
x^3-3x=\zeta_7^3+\zeta_7^{-3}$を用いれば
\[
x^3+x^2-2x-1
\]
が求める最小多項式である．この最小多項式の
方程式に$\sigma_3$を作用させることにより
\[
x^3+x^2-2x-1=(x-2\mathrm{cos}\frac{2\pi}{7})
(x-2\mathrm{cos}\frac{4\pi}{7})
(x-2\mathrm{cos}\frac{6\pi}{7})
\]
がわかる．
\begin{theorem}
    $(\mathbb{Z}/m\mathbb{Z})^\times$
    が巡回群であることと
    $m=1,2,4,p^k,2p^k,\,(k\in\mathbb{N},p\colon \text{奇素数})$
    は同値である．(hardy and wright "An Introduction to the 
    Theory of Numbers")
\end{theorem}

\begin{theorem}\label{dirichlet}
    初項と公差が互いに素な等差数列には無限に素数が存在する．
    （ディリクレの算術級数定理）
\end{theorem}
任意の素数$p$に対して初項を$1$，公差を$p$とすればディリクレの算術級数
定理\uref{dirichlet}より
ある素数$q$が存在する．このとき$p\mid q-1$である．
これを用いると任意の$p$に対して$\mathbb{Q}$の
$p$次巡回拡大を$q-1$次のガロア拡大
$\mathbb{Q}(\zeta_q)/\mathbb{Q}$
の部分体で構成できる．

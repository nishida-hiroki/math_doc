\section*{半直積}

\begin{definition}[内部半直積]
    群$G$の部分群$N,H$が次を満たすとき，
    $G$は$N$と$H$の\textbf{（内部）半直積}であるといい，
    $G=N\rtimes H$と書く．
    \begin{enumerate}
        \item $N\trianglelefteq G$（$N$は$G$の正規部分群）
        \item $G=NH$
        \item $N\cap H=\{e\}$
    \end{enumerate}
\end{definition}

\begin{definition}[外部半直積]
    群$N,H$と群準同型$\phi\colon H\to\mathrm{Aut}(N)$が与えられたとき，
    集合$N\times H$に積を
    $$(n_1,h_1)(n_2,h_2)=(n_1\,\phi(h_1)(n_2),\,h_1h_2)$$
    で定めると群になる．
    この群を$N,H$の$\phi$による\textbf{（外部）半直積}といい，
    $N\rtimes_\phi H$と書く．
\end{definition}

\begin{remark}
    内部半直積と外部半直積は次の意味で一致する．
    $G=N\rtimes H$（内部）のとき，
    $\phi\colon H\to\mathrm{Aut}(N)$を
    $\phi(h)(n)=hnh^{-1}$で定めると
    $G\cong N\rtimes_\phi H$（外部）となる．
    逆に外部半直積$N\rtimes_\phi H$において，
    $\bar{N}=\{(n,e)\mid n\in N\}$，
    $\bar{H}=\{(e,h)\mid h\in H\}$
    とおけば$N\rtimes_\phi H$は
    $\bar{N}$と$\bar{H}$の内部半直積である．
\end{remark}

\section*{群拡大}

\begin{definition}[群拡大]
    群$N,G,Q$と群準同型$i\colon N\to G$，$\pi\colon G\to Q$が
    \begin{equation*}
        1\longrightarrow N\xrightarrow{\;i\;} G\xrightarrow{\;\pi\;} Q\longrightarrow 1
    \end{equation*}
    という完全列をなすとき，$G$を$N$による$Q$の\textbf{群拡大}という．
    すなわち$i$は単射，$\pi$は全射，$\mathrm{Im}(i)=\ker(\pi)$が成り立つ．
    $N$を\textbf{核}，$Q$を\textbf{商}と呼ぶ．
\end{definition}

\begin{definition}[分裂拡大]
    群拡大
    $1\to N\xrightarrow{i} G\xrightarrow{\pi} Q\to 1$
    において，群準同型$s\colon Q\to G$で
    $\pi\circ s=\mathrm{id}_Q$を満たすものが存在するとき，
    この拡大は\textbf{分裂する}という．
    $s$を\textbf{断面（section）}と呼ぶ．
\end{definition}

\begin{proposition}
    群拡大$1\to N\xrightarrow{i} G\xrightarrow{\pi} Q\to 1$が
    分裂するならば，$G\cong N\rtimes_\phi Q$（半直積）である．
    ここで$\phi\colon Q\to\mathrm{Aut}(N)$は
    $\phi(q)(n)=s(q)\,i(n)\,s(q)^{-1}$で定まる．
\end{proposition}

\begin{remark}
    分裂しない群拡大の例として，$\mathbb{Z}/4\mathbb{Z}$は
    $\mathbb{Z}/2\mathbb{Z}$による$\mathbb{Z}/2\mathbb{Z}$の拡大
    $1\to\mathbb{Z}/2\mathbb{Z}\to\mathbb{Z}/4\mathbb{Z}\to\mathbb{Z}/2\mathbb{Z}\to 1$
    を与えるが，断面が群準同型にならないため分裂しない．
    一方$S_3\cong\mathbb{Z}/3\mathbb{Z}\rtimes\mathbb{Z}/2\mathbb{Z}$は分裂する拡大の例である．
\end{remark}


\section*{位相と収束}

$(X,\mathcal{M},\mu)$を測度空間とし，$f_n,f\colon X\to\mathbb{R}$を可測関数の列とする．

\begin{definition}[位相空間での収束]
    位相空間$(X,\mathcal{O})$において，点列$\{x_n\}_{n=1}^\infty\subset X$が
    $x\in X$に\textbf{収束}するとは，
    $x$の任意の開近傍$U$（$x\in U\in\mathcal{O}$）に対して，
    ある$N\in\mathbb{N}$が存在して
    $$n\geq N\Rightarrow x_n\in U$$
    が成り立つことをいう．このとき$x_n\to x$と書く．
\end{definition}

\begin{remark}
    Hausdorff空間（$T_2$空間）では極限は一意である．
    一般の位相空間では極限が一意でないことがある．
    また，位相空間では点列収束だけでは位相を特徴づけられないことがあり，
    その場合はネット（有向集合で添え字付けられた点族）や
    フィルターによる収束を用いる．
\end{remark}

\begin{definition}[ノルム位相]
    ノルム空間$(V,\|\cdot\|)$において，
    開球$B(x,r)=\{y\in V\mid \|y-x\|<r\}$を基底とする位相を
    $V$の\textbf{ノルム位相}という．
\end{definition}

\begin{definition}[一様ノルム位相]
    集合$X$上の有界関数全体$B(X)$に一様ノルム
    $\|f\|_\infty=\sup_{x\in X}|f(x)|$
    を入れたとき，このノルムが定めるノルム位相を
    $B(X)$の\textbf{一様ノルム位相}（一様収束位相）という．
\end{definition}

\begin{definition}[$L^p$位相 ($1\leq p\leq\infty$)]
    測度空間$(X,\mathcal{M},\mu)$上の$L^p$空間に
    $L^p$ノルム$\|\cdot\|_{L^p}$が定めるノルム位相を
    $L^p(X,\mu)$の\textbf{$L^p$位相}という．
\end{definition}

\begin{definition}[積位相]
    位相空間の族$\{(X_\lambda,\mathcal{O}_\lambda)\}_{\lambda\in\Lambda}$に対し，
    直積集合$\prod_{\lambda\in\Lambda}X_\lambda$上の位相で，
    各射影$\pi_\lambda\colon\prod X_\lambda\to X_\lambda$が連続となる
    最も粗い位相を\textbf{積位相}という．
    開基は有限個の$\lambda$を除いて$X_\lambda$となる集合の族
    $$\left\{\prod_{\lambda\in\Lambda}U_\lambda\ \middle|\
    U_\lambda\in\mathcal{O}_\lambda,\,
    U_\lambda=X_\lambda \text{ (有限個を除く)}\right\}$$
    で与えられる．
\end{definition}

\begin{definition}[各点収束]
    任意の$x\in X$に対して
    $$\lim_{n\to\infty}f_n(x)=f(x)$$
    が成り立つとき，$f_n$は$f$に\textbf{各点収束}するという．
\end{definition}

\begin{definition}[一様収束]
    $$\lim_{n\to\infty}\sup_{x\in X}|f_n(x)-f(x)|=0$$
    が成り立つとき，$f_n$は$f$に\textbf{一様収束}するという．
\end{definition}

\begin{definition}[$L^\infty$収束]
    $f_n,f\in L^\infty(X,\mu)$とする．
    $$\lim_{n\to\infty}\|f_n-f\|_{L^\infty}=0,
    \quad \|g\|_{L^\infty}=\mathrm{ess\,sup}_{x\in X}|g(x)|$$
    が成り立つとき，$f_n$は$f$に$\bm{L^\infty}$\textbf{収束}するという．
    ここで本質的上限は
    $$\mathrm{ess\,sup}_{x\in X}|g(x)|
    =\inf\{M\geq 0\mid \mu(\{x\mid |g(x)|>M\})=0\}$$
    である．
\end{definition}

\begin{definition}[$L^p$収束 ($1\leq p<\infty$)]
    $f_n,f\in L^p(X,\mu)$とする．
    $$\lim_{n\to\infty}\|f_n-f\|_{L^p}=0,
    \quad \|g\|_{L^p}=\left(\int_X|g|^p\,d\mu\right)^{1/p}$$
    が成り立つとき，$f_n$は$f$に$\bm{L^p}$\textbf{収束}するという．
\end{definition}

\begin{definition}[概収束 (a.e.収束)]
    $\mu(\{x\in X\mid f_n(x)\not\to f(x)\})=0$
    が成り立つとき，すなわち「ほとんどすべての$x\in X$で
    $\lim_{n\to\infty}f_n(x)=f(x)$」が成り立つとき，
    $f_n$は$f$に\textbf{概収束}（\textbf{a.e.収束}）するという．
    記号で$f_n\to f\ \text{a.e.}$と書く．
\end{definition}

\begin{definition}[測度収束]
    任意の$\varepsilon>0$に対して
    $$\lim_{n\to\infty}\mu(\{x\in X\mid |f_n(x)-f(x)|>\varepsilon\})=0$$
    が成り立つとき，$f_n$は$f$に\textbf{測度収束}するという．
\end{definition}


\section*{距離関数}

\begin{definition}[コサイン類似度・コサイン距離]
    $\bm{x},\bm{y}\in\mathbb{R}^n\setminus\{\bm{0}\}$に対して，
    \textbf{コサイン類似度}を
    $$\cos(\bm{x},\bm{y})
    =\frac{\bm{x}\cdot\bm{y}}{\|\bm{x}\|\,\|\bm{y}\|}
    =\frac{\sum_{i=1}^n x_i y_i}
    {\sqrt{\sum_{i=1}^n x_i^2}\,\sqrt{\sum_{i=1}^n y_i^2}}$$
    で定義する．値域は$[-1,1]$であり，
    $\bm{x}$と$\bm{y}$のなす角を$\theta$とすると
    $\cos(\bm{x},\bm{y})=\cos\theta$に等しい．\\
    \textbf{コサイン距離}を
    $$d_{\cos}(\bm{x},\bm{y})=1-\cos(\bm{x},\bm{y})$$
    で定義する．値域は$[0,2]$であり，
    $\bm{x},\bm{y}$が同方向のとき$0$，逆方向のとき$2$となる．
\end{definition}

\begin{remark}
    コサイン距離は距離の公理のうち，\textbf{非退化性}と\textbf{三角不等式}を満たさない．
    \begin{enumerate}
        \item \textbf{非退化性の破れ}：
        距離の公理では$d(\bm{x},\bm{y})=0\Leftrightarrow\bm{x}=\bm{y}$を要求するが，
        コサイン距離では
        $$d_{\cos}(\bm{x},\bm{y})=0
        \iff\cos(\bm{x},\bm{y})=1
        \iff\bm{x}=\lambda\bm{y}\text{ (ある}\lambda>0\text{)}$$
        となる．すなわち同方向のスカラー倍であればよく，
        $\bm{x}\neq\bm{y}$であっても$d_{\cos}(\bm{x},\bm{y})=0$になりうる．
        これはコサイン距離がベクトルの\textbf{向き}のみを比較し，
        大きさを無視することによる．
        \item \textbf{三角不等式の破れ}：
        $\mathbb{R}^2$において
        $\bm{x}=(1,0)$，
        $\bm{y}=\tfrac{1}{\sqrt{2}}(1,1)$，
        $\bm{z}=(0,1)$とおくと，
        \begin{align*}
            d_{\cos}(\bm{x},\bm{y})&=1-\tfrac{1}{\sqrt{2}}\approx 0.293,\\
            d_{\cos}(\bm{y},\bm{z})&=1-\tfrac{1}{\sqrt{2}}\approx 0.293,\\
            d_{\cos}(\bm{x},\bm{z})&=1-0=1
        \end{align*}
        であるから，
        $d_{\cos}(\bm{x},\bm{z})=1>0.586\approx
        d_{\cos}(\bm{x},\bm{y})+d_{\cos}(\bm{y},\bm{z})$
        となり三角不等式が成り立たない．
        これは$\bm{x},\bm{y},\bm{z}$の角度が等差的に並んでいるとき，
        距離が線形でなく余弦を通じて定義されることが原因である．
    \end{enumerate}
    以上より$d_{\cos}$は厳密には距離関数ではなく\textbf{非類似度}である．
    自然言語処理や情報検索においてベクトルの向きの類似性を測るために広く用いられる．
\end{remark}

\begin{definition}[正定値対称行列]
    $A\in M_n(\mathbb{R})$が\textbf{対称行列}であるとは$\prescript{t}{}{}A=A$が成り立つことをいう．
    対称行列$A$が\textbf{正定値}であるとは，すべての零でないベクトル$\bm{x}\in\mathbb{R}^n\setminus\{\bm{0}\}$に対して
    $$\prescript{t}{}{}\bm{x}A\bm{x}>0$$
    が成り立つことをいう．このような行列を\textbf{正定値対称行列}という．
    等号も許す条件$\prescript{t}{}{}\bm{x}A\bm{x}\geq 0$（$\bm{x}\neq\bm{0}$）のとき\textbf{半正定値}という．
\end{definition}

\begin{remark}
    正定値対称行列$A$は次の同値な条件をもつ：
    \begin{enumerate}
        \item すべての固有値が正である．
        \item コレスキー分解$A=\prescript{t}{}{}B\,B$（$B$：正則行列）が存在する．
        \item $A$は正則であり，$A^{-1}$も正定値対称行列である．
    \end{enumerate}
\end{remark}

\begin{definition}[共分散行列]
    $\bm{X}=\prescript{t}{}{}(X_1,\ldots,X_n)$を$\mathbb{R}^n$値確率変数とし，
    $\bm{\mu}=E[\bm{X}]\in\mathbb{R}^n$をその期待値とする．
    \textbf{共分散行列}（分散共分散行列）$\Sigma\in M_n(\mathbb{R})$を
    $$\Sigma=E\bigl[(\bm{X}-\bm{\mu})\,\prescript{t}{}{}(\bm{X}-\bm{\mu})\bigr]$$
    で定義する．成分で書くと$(i,j)$成分は
    $$\Sigma_{ij}=\mathrm{Cov}(X_i,X_j)=E[(X_i-\mu_i)(X_j-\mu_j)]$$
    であり，特に対角成分$\Sigma_{ii}=\mathrm{Var}(X_i)$は各成分の分散である．
\end{definition}

\begin{remark}
    共分散行列は常に\textbf{半正定値対称行列}である．
    すなわち$\prescript{t}{}{}\Sigma=\Sigma$かつ任意の$\bm{v}\in\mathbb{R}^n$に対して
    $\prescript{t}{}{}\bm{v}\Sigma\bm{v}=\mathrm{Var}(\prescript{t}{}{}\bm{v}\bm{X})\geq 0$が成り立つ．
    $\bm{X}$の成分の間に完全な線形従属関係がないとき（非退化のとき），$\Sigma$は正定値対称行列になる．
\end{remark}

\begin{definition}[マハラノビス距離]
    $\bm{x},\bm{y}\in\mathbb{R}^n$と正定値対称行列$\Sigma\in M_n(\mathbb{R})$に対して，
    \textbf{マハラノビス距離}を
    $$d_M(\bm{x},\bm{y})=\sqrt{\prescript{t}{}{}(\bm{x}-\bm{y})\Sigma^{-1}(\bm{x}-\bm{y})}$$
    で定義する．
    特に確率分布$P$を考えるとき，$\Sigma$として$P$の共分散行列をとることが多い．
    $\Sigma=I$（単位行列）のときユークリッド距離に一致する．
\end{definition}

\begin{remark}
    マハラノビス距離はデータの\textbf{スケール}と\textbf{相関}を考慮した距離である．
    各成分の分散が異なる場合や成分間に相関がある場合に，
    ユークリッド距離より適切な距離を与える．
    \begin{enumerate}
        \item \textbf{スケール不変性}：
        各変数を標準化した場合でも同じ値を与える．
        すなわち$\bm{x}$の第$i$成分を$\sigma_i$倍しても距離は変わらない．
        \item \textbf{相関の補正}：
        $\Sigma^{-1}$によって変数間の相関を取り除いた空間でのユークリッド距離に相当する．
        より正確には，$\Sigma=A\,\prescript{t}{}{}A$と分解するとき（Cholesky分解など），
        $d_M(\bm{x},\bm{y})=\|A^{-1}(\bm{x}-\bm{y})\|$である．
        \item \textbf{距離の公理}：
        $\Sigma$が正定値対称行列ならば，$d_M$は距離の公理をすべて満たす．
    \end{enumerate}
\end{remark}

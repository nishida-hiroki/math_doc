\section*{半直積}

\begin{definition}[内部半直積]
    群$G$の部分群$N,H$が次を満たすとき，
    $G$は$N$と$H$の\textbf{（内部）半直積}であるといい，
    $G=N\rtimes H$と書く．
    \begin{enumerate}
        \item $N\trianglelefteq G$（$N$は$G$の正規部分群）
        \item $G=NH$
        \item $N\cap H=\{e\}$
    \end{enumerate}
\end{definition}

\begin{definition}[外部半直積]
    群$N,H$と群準同型$\phi\colon H\to\mathrm{Aut}(N)$が与えられたとき，
    集合$N\times H$に積を
    $$(n_1,h_1)(n_2,h_2)=(n_1\,\phi(h_1)(n_2),\,h_1h_2)$$
    で定めると群になる．
    この群を$N,H$の$\phi$による\textbf{（外部）半直積}といい，
    $N\rtimes_\phi H$と書く．
\end{definition}

\begin{remark}
    内部半直積と外部半直積は次の意味で一致する．
    $G=N\rtimes H$（内部）のとき，
    $\phi\colon H\to\mathrm{Aut}(N)$を
    $\phi(h)(n)=hnh^{-1}$で定めると
    $G\cong N\rtimes_\phi H$（外部）となる．
    逆に外部半直積$N\rtimes_\phi H$において，
    $\bar{N}=\{(n,e)\mid n\in N\}$，
    $\bar{H}=\{(e,h)\mid h\in H\}$
    とおけば$N\rtimes_\phi H$は
    $\bar{N}$と$\bar{H}$の内部半直積である．
\end{remark}

\section*{群拡大}

\begin{definition}[群拡大]
    群$N,G,Q$と群準同型$i\colon N\to G$，$\pi\colon G\to Q$が
    \begin{equation*}
        1\longrightarrow N\xrightarrow{\;i\;} G\xrightarrow{\;\pi\;} Q\longrightarrow 1
    \end{equation*}
    という完全列をなすとき，$G$を$N$による$Q$の\textbf{群拡大}という．
    すなわち$i$は単射，$\pi$は全射，$\mathrm{Im}(i)=\ker(\pi)$が成り立つ．
    $N$を\textbf{核}，$Q$を\textbf{商}と呼ぶ．
\end{definition}

\begin{definition}[分裂拡大]
    群拡大
    $1\to N\xrightarrow{i} G\xrightarrow{\pi} Q\to 1$
    において，群準同型$s\colon Q\to G$で
    $\pi\circ s=\mathrm{id}_Q$を満たすものが存在するとき，
    この拡大は\textbf{分裂する}という．
    $s$を\textbf{断面（section）}と呼ぶ．
\end{definition}

\begin{proposition}
    群拡大$1\to N\xrightarrow{i} G\xrightarrow{\pi} Q\to 1$が
    分裂するならば，$G\cong N\rtimes_\phi Q$（半直積）である．
    ここで$\phi\colon Q\to\mathrm{Aut}(N)$は
    $\phi(q)(n)=s(q)\,i(n)\,s(q)^{-1}$で定まる．
\end{proposition}

\begin{remark}
    分裂しない群拡大の例として，$\mathbb{Z}/4\mathbb{Z}$は
    $\mathbb{Z}/2\mathbb{Z}$による$\mathbb{Z}/2\mathbb{Z}$の拡大
    $1\to\mathbb{Z}/2\mathbb{Z}\to\mathbb{Z}/4\mathbb{Z}\to\mathbb{Z}/2\mathbb{Z}\to 1$
    を与えるが，断面が群準同型にならないため分裂しない．
    一方$S_3\cong\mathbb{Z}/3\mathbb{Z}\rtimes\mathbb{Z}/2\mathbb{Z}$は分裂する拡大の例である．
\end{remark}


\section*{位相と収束}

$(X,\mathcal{M},\mu)$を測度空間とし，$f_n,f\colon X\to\mathbb{R}$を可測関数の列とする．

\begin{definition}[位相空間での収束]
    位相空間$(X,\mathcal{O})$において，点列$\{x_n\}_{n=1}^\infty\subset X$が
    $x\in X$に\textbf{収束}するとは，
    $x$の任意の開近傍$U$（$x\in U\in\mathcal{O}$）に対して，
    ある$N\in\mathbb{N}$が存在して
    $$n\geq N\Rightarrow x_n\in U$$
    が成り立つことをいう．このとき$x_n\to x$と書く．
\end{definition}

\begin{remark}
    Hausdorff空間（$T_2$空間）では極限は一意である．
    一般の位相空間では極限が一意でないことがある．
    また，位相空間では点列収束だけでは位相を特徴づけられないことがあり，
    その場合はネット（有向集合で添え字付けられた点族）や
    フィルターによる収束を用いる．
\end{remark}

\begin{definition}[ノルム位相]
    ノルム空間$(V,\|\cdot\|)$において，
    開球$B(x,r)=\{y\in V\mid \|y-x\|<r\}$を基底とする位相を
    $V$の\textbf{ノルム位相}という．
\end{definition}

\begin{definition}[一様ノルム位相]
    集合$X$上の有界関数全体$B(X)$に一様ノルム
    $\|f\|_\infty=\sup_{x\in X}|f(x)|$
    を入れたとき，このノルムが定めるノルム位相を
    $B(X)$の\textbf{一様ノルム位相}（一様収束位相）という．
\end{definition}

\begin{definition}[$L^p$位相 ($1\leq p\leq\infty$)]
    測度空間$(X,\mathcal{M},\mu)$上の$L^p$空間に
    $L^p$ノルム$\|\cdot\|_{L^p}$が定めるノルム位相を
    $L^p(X,\mu)$の\textbf{$L^p$位相}という．
\end{definition}

\begin{definition}[積位相]
    位相空間の族$\{(X_\lambda,\mathcal{O}_\lambda)\}_{\lambda\in\Lambda}$に対し，
    直積集合$\prod_{\lambda\in\Lambda}X_\lambda$上の位相で，
    各射影$\pi_\lambda\colon\prod X_\lambda\to X_\lambda$が連続となる
    最も粗い位相を\textbf{積位相}という．
    開基は有限個の$\lambda$を除いて$X_\lambda$となる集合の族
    $$\left\{\prod_{\lambda\in\Lambda}U_\lambda\ \middle|\
    U_\lambda\in\mathcal{O}_\lambda,\,
    U_\lambda=X_\lambda \text{ (有限個を除く)}\right\}$$
    で与えられる．
\end{definition}

\begin{definition}[各点収束]
    任意の$x\in X$に対して
    $$\lim_{n\to\infty}f_n(x)=f(x)$$
    が成り立つとき，$f_n$は$f$に\textbf{各点収束}するという．
\end{definition}

\begin{definition}[一様収束]
    $$\lim_{n\to\infty}\sup_{x\in X}|f_n(x)-f(x)|=0$$
    が成り立つとき，$f_n$は$f$に\textbf{一様収束}するという．
\end{definition}

\begin{definition}[$L^\infty$収束]
    $f_n,f\in L^\infty(X,\mu)$とする．
    $$\lim_{n\to\infty}\|f_n-f\|_{L^\infty}=0,
    \quad \|g\|_{L^\infty}=\mathrm{ess\,sup}_{x\in X}|g(x)|$$
    が成り立つとき，$f_n$は$f$に$\bm{L^\infty}$\textbf{収束}するという．
    ここで本質的上限は
    $$\mathrm{ess\,sup}_{x\in X}|g(x)|
    =\inf\{M\geq 0\mid \mu(\{x\mid |g(x)|>M\})=0\}$$
    である．
\end{definition}

\begin{definition}[$L^p$収束 ($1\leq p<\infty$)]
    $f_n,f\in L^p(X,\mu)$とする．
    $$\lim_{n\to\infty}\|f_n-f\|_{L^p}=0,
    \quad \|g\|_{L^p}=\left(\int_X|g|^p\,d\mu\right)^{1/p}$$
    が成り立つとき，$f_n$は$f$に$\bm{L^p}$\textbf{収束}するという．
\end{definition}

\begin{definition}[概収束 (a.e.収束)]
    $\mu(\{x\in X\mid f_n(x)\not\to f(x)\})=0$
    が成り立つとき，すなわち「ほとんどすべての$x\in X$で
    $\lim_{n\to\infty}f_n(x)=f(x)$」が成り立つとき，
    $f_n$は$f$に\textbf{概収束}（\textbf{a.e.収束}）するという．
    記号で$f_n\to f\ \text{a.e.}$と書く．
\end{definition}

\begin{definition}[測度収束]
    任意の$\varepsilon>0$に対して
    $$\lim_{n\to\infty}\mu(\{x\in X\mid |f_n(x)-f(x)|>\varepsilon\})=0$$
    が成り立つとき，$f_n$は$f$に\textbf{測度収束}するという．
\end{definition}

\section*{半直積}

\begin{definition}[内部半直積]
    群$G$の部分群$N,H$が次を満たすとき，
    $G$は$N$と$H$の\textbf{（内部）半直積}であるといい，
    $G=N\rtimes H$と書く．
    \begin{enumerate}
        \item $N\trianglelefteq G$（$N$は$G$の正規部分群）
        \item $G=NH$
        \item $N\cap H=\{e\}$
    \end{enumerate}
\end{definition}

\begin{definition}[外部半直積]
    群$N,H$と群準同型$\phi\colon H\to\mathrm{Aut}(N)$が与えられたとき，
    集合$N\times H$に積を
    $$(n_1,h_1)(n_2,h_2)=(n_1\,\phi(h_1)(n_2),\,h_1h_2)$$
    で定めると群になる．
    この群を$N,H$の$\phi$による\textbf{（外部）半直積}といい，
    $N\rtimes_\phi H$と書く．
\end{definition}

\begin{remark}
    内部半直積と外部半直積は次の意味で一致する．
    $G=N\rtimes H$（内部）のとき，
    $\phi\colon H\to\mathrm{Aut}(N)$を
    $\phi(h)(n)=hnh^{-1}$で定めると
    $G\cong N\rtimes_\phi H$（外部）となる．
    逆に外部半直積$N\rtimes_\phi H$において，
    $\bar{N}=\{(n,e)\mid n\in N\}$，
    $\bar{H}=\{(e,h)\mid h\in H\}$
    とおけば$N\rtimes_\phi H$は
    $\bar{N}$と$\bar{H}$の内部半直積である．
\end{remark}

\section*{群拡大}

\begin{definition}[群拡大]
    群$N,G,Q$と群準同型$i\colon N\to G$，$\pi\colon G\to Q$が
    \begin{equation*}
        1\longrightarrow N\xrightarrow{\;i\;} G\xrightarrow{\;\pi\;} Q\longrightarrow 1
    \end{equation*}
    という完全列をなすとき，$G$を$N$による$Q$の\textbf{群拡大}という．
    すなわち$i$は単射，$\pi$は全射，$\mathrm{Im}(i)=\ker(\pi)$が成り立つ．
    $N$を\textbf{核}，$Q$を\textbf{商}と呼ぶ．
\end{definition}

\begin{definition}[分裂拡大]
    群拡大
    $1\to N\xrightarrow{i} G\xrightarrow{\pi} Q\to 1$
    において，群準同型$s\colon Q\to G$で
    $\pi\circ s=\mathrm{id}_Q$を満たすものが存在するとき，
    この拡大は\textbf{分裂する}という．
    $s$を\textbf{断面（section）}と呼ぶ．
\end{definition}

\begin{proposition}
    群拡大$1\to N\xrightarrow{i} G\xrightarrow{\pi} Q\to 1$が
    分裂するならば，$G\cong N\rtimes_\phi Q$（半直積）である．
    ここで$\phi\colon Q\to\mathrm{Aut}(N)$は
    $\phi(q)(n)=s(q)\,i(n)\,s(q)^{-1}$で定まる．
\end{proposition}

\begin{remark}
    分裂しない群拡大の例として，$\mathbb{Z}/4\mathbb{Z}$は
    $\mathbb{Z}/2\mathbb{Z}$による$\mathbb{Z}/2\mathbb{Z}$の拡大
    $1\to\mathbb{Z}/2\mathbb{Z}\to\mathbb{Z}/4\mathbb{Z}\to\mathbb{Z}/2\mathbb{Z}\to 1$
    を与えるが，断面が群準同型にならないため分裂しない．
    一方$S_3\cong\mathbb{Z}/3\mathbb{Z}\rtimes\mathbb{Z}/2\mathbb{Z}$は分裂する拡大の例である．
\end{remark}


\section*{位相と収束}

$(X,\mathcal{M},\mu)$を測度空間とし，$f_n,f\colon X\to\mathbb{R}$を可測関数の列とする．

\begin{definition}[位相空間での収束]
    位相空間$(X,\mathcal{O})$において，点列$\{x_n\}_{n=1}^\infty\subset X$が
    $x\in X$に\textbf{収束}するとは，
    $x$の任意の開近傍$U$（$x\in U\in\mathcal{O}$）に対して，
    ある$N\in\mathbb{N}$が存在して
    $$n\geq N\Rightarrow x_n\in U$$
    が成り立つことをいう．このとき$x_n\to x$と書く．
\end{definition}

\begin{remark}
    Hausdorff空間（$T_2$空間）では極限は一意である．
    一般の位相空間では極限が一意でないことがある．
    また，位相空間では点列収束だけでは位相を特徴づけられないことがあり，
    その場合はネット（有向集合で添え字付けられた点族）や
    フィルターによる収束を用いる．
\end{remark}

\begin{definition}[ノルム位相]
    ノルム空間$(V,\|\cdot\|)$において，
    開球$B(x,r)=\{y\in V\mid \|y-x\|<r\}$を基底とする位相を
    $V$の\textbf{ノルム位相}という．
\end{definition}

\begin{definition}[一様ノルム位相]
    集合$X$上の有界関数全体$B(X)$に一様ノルム
    $\|f\|_\infty=\sup_{x\in X}|f(x)|$
    を入れたとき，このノルムが定めるノルム位相を
    $B(X)$の\textbf{一様ノルム位相}（一様収束位相）という．
\end{definition}

\begin{definition}[$L^p$位相 ($1\leq p\leq\infty$)]
    測度空間$(X,\mathcal{M},\mu)$上の$L^p$空間に
    $L^p$ノルム$\|\cdot\|_{L^p}$が定めるノルム位相を
    $L^p(X,\mu)$の\textbf{$L^p$位相}という．
\end{definition}

\begin{definition}[積位相]
    位相空間の族$\{(X_\lambda,\mathcal{O}_\lambda)\}_{\lambda\in\Lambda}$に対し，
    直積集合$\prod_{\lambda\in\Lambda}X_\lambda$上の位相で，
    各射影$\pi_\lambda\colon\prod X_\lambda\to X_\lambda$が連続となる
    最も粗い位相を\textbf{積位相}という．
    開基は有限個の$\lambda$を除いて$X_\lambda$となる集合の族
    $$\left\{\prod_{\lambda\in\Lambda}U_\lambda\ \middle|\
    U_\lambda\in\mathcal{O}_\lambda,\,
    U_\lambda=X_\lambda \text{ (有限個を除く)}\right\}$$
    で与えられる．
\end{definition}

\begin{definition}[各点収束]
    任意の$x\in X$に対して
    $$\lim_{n\to\infty}f_n(x)=f(x)$$
    が成り立つとき，$f_n$は$f$に\textbf{各点収束}するという．
\end{definition}

\begin{definition}[一様収束]
    $$\lim_{n\to\infty}\sup_{x\in X}|f_n(x)-f(x)|=0$$
    が成り立つとき，$f_n$は$f$に\textbf{一様収束}するという．
\end{definition}

\begin{definition}[$L^\infty$収束]
    $f_n,f\in L^\infty(X,\mu)$とする．
    $$\lim_{n\to\infty}\|f_n-f\|_{L^\infty}=0,
    \quad \|g\|_{L^\infty}=\mathrm{ess\,sup}_{x\in X}|g(x)|$$
    が成り立つとき，$f_n$は$f$に$\bm{L^\infty}$\textbf{収束}するという．
    ここで本質的上限は
    $$\mathrm{ess\,sup}_{x\in X}|g(x)|
    =\inf\{M\geq 0\mid \mu(\{x\mid |g(x)|>M\})=0\}$$
    である．
\end{definition}

\begin{definition}[$L^p$収束 ($1\leq p<\infty$)]
    $f_n,f\in L^p(X,\mu)$とする．
    $$\lim_{n\to\infty}\|f_n-f\|_{L^p}=0,
    \quad \|g\|_{L^p}=\left(\int_X|g|^p\,d\mu\right)^{1/p}$$
    が成り立つとき，$f_n$は$f$に$\bm{L^p}$\textbf{収束}するという．
\end{definition}

\begin{definition}[概収束 (a.e.収束)]
    $\mu(\{x\in X\mid f_n(x)\not\to f(x)\})=0$
    が成り立つとき，すなわち「ほとんどすべての$x\in X$で
    $\lim_{n\to\infty}f_n(x)=f(x)$」が成り立つとき，
    $f_n$は$f$に\textbf{概収束}（\textbf{a.e.収束}）するという．
    記号で$f_n\to f\ \text{a.e.}$と書く．
\end{definition}

\begin{definition}[測度収束]
    任意の$\varepsilon>0$に対して
    $$\lim_{n\to\infty}\mu(\{x\in X\mid |f_n(x)-f(x)|>\varepsilon\})=0$$
    が成り立つとき，$f_n$は$f$に\textbf{測度収束}するという．
\end{definition}


\section*{距離関数}

\begin{definition}[コサイン類似度・コサイン距離]
    $\bm{x},\bm{y}\in\mathbb{R}^n\setminus\{\bm{0}\}$に対して，
    \textbf{コサイン類似度}を
    $$\cos(\bm{x},\bm{y})
    =\frac{\bm{x}\cdot\bm{y}}{\|\bm{x}\|\,\|\bm{y}\|}
    =\frac{\sum_{i=1}^n x_i y_i}
    {\sqrt{\sum_{i=1}^n x_i^2}\,\sqrt{\sum_{i=1}^n y_i^2}}$$
    で定義する．値域は$[-1,1]$であり，
    $\bm{x}$と$\bm{y}$のなす角を$\theta$とすると
    $\cos(\bm{x},\bm{y})=\cos\theta$に等しい．\\
    \textbf{コサイン距離}を
    $$d_{\cos}(\bm{x},\bm{y})=1-\cos(\bm{x},\bm{y})$$
    で定義する．値域は$[0,2]$であり，
    $\bm{x},\bm{y}$が同方向のとき$0$，逆方向のとき$2$となる．
\end{definition}

\begin{remark}
    コサイン距離は距離の公理のうち，\textbf{非退化性}と\textbf{三角不等式}を満たさない．
    \begin{enumerate}
        \item \textbf{非退化性の破れ}：
        距離の公理では$d(\bm{x},\bm{y})=0\Leftrightarrow\bm{x}=\bm{y}$を要求するが，
        コサイン距離では
        $$d_{\cos}(\bm{x},\bm{y})=0
        \iff\cos(\bm{x},\bm{y})=1
        \iff\bm{x}=\lambda\bm{y}\text{ (ある}\lambda>0\text{)}$$
        となる．すなわち同方向のスカラー倍であればよく，
        $\bm{x}\neq\bm{y}$であっても$d_{\cos}(\bm{x},\bm{y})=0$になりうる．
        これはコサイン距離がベクトルの\textbf{向き}のみを比較し，
        大きさを無視することによる．
        \item \textbf{三角不等式の破れ}：
        $\mathbb{R}^2$において
        $\bm{x}=(1,0)$，
        $\bm{y}=\tfrac{1}{\sqrt{2}}(1,1)$，
        $\bm{z}=(0,1)$とおくと，
        \begin{align*}
            d_{\cos}(\bm{x},\bm{y})&=1-\tfrac{1}{\sqrt{2}}\approx 0.293,\\
            d_{\cos}(\bm{y},\bm{z})&=1-\tfrac{1}{\sqrt{2}}\approx 0.293,\\
            d_{\cos}(\bm{x},\bm{z})&=1-0=1
        \end{align*}
        であるから，
        $d_{\cos}(\bm{x},\bm{z})=1>0.586\approx
        d_{\cos}(\bm{x},\bm{y})+d_{\cos}(\bm{y},\bm{z})$
        となり三角不等式が成り立たない．
        これは$\bm{x},\bm{y},\bm{z}$の角度が等差的に並んでいるとき，
        距離が線形でなく余弦を通じて定義されることが原因である．
    \end{enumerate}
    以上より$d_{\cos}$は厳密には距離関数ではなく\textbf{非類似度}である．
    自然言語処理や情報検索においてベクトルの向きの類似性を測るために広く用いられる．
\end{remark}

\begin{definition}[正定値対称行列]
    $A\in M_n(\mathbb{R})$が\textbf{対称行列}であるとは$\prescript{t}{}{}A=A$が成り立つことをいう．
    対称行列$A$が\textbf{正定値}であるとは，すべての零でないベクトル$\bm{x}\in\mathbb{R}^n\setminus\{\bm{0}\}$に対して
    $$\prescript{t}{}{}\bm{x}A\bm{x}>0$$
    が成り立つことをいう．このような行列を\textbf{正定値対称行列}という．
    等号も許す条件$\prescript{t}{}{}\bm{x}A\bm{x}\geq 0$（$\bm{x}\neq\bm{0}$）のとき\textbf{半正定値}という．
\end{definition}

\begin{remark}
    正定値対称行列$A$は次の同値な条件をもつ：
    \begin{enumerate}
        \item すべての固有値が正である．
        \item コレスキー分解$A=\prescript{t}{}{}B\,B$（$B$：正則行列）が存在する．
        \item $A$は正則であり，$A^{-1}$も正定値対称行列である．
    \end{enumerate}
\end{remark}

\begin{definition}[共分散行列]
    $\bm{X}=\prescript{t}{}{}(X_1,\ldots,X_n)$を$\mathbb{R}^n$値確率変数とし，
    $\bm{\mu}=E[\bm{X}]\in\mathbb{R}^n$をその期待値とする．
    \textbf{共分散行列}（分散共分散行列）$\Sigma\in M_n(\mathbb{R})$を
    $$\Sigma=E\bigl[(\bm{X}-\bm{\mu})\,\prescript{t}{}{}(\bm{X}-\bm{\mu})\bigr]$$
    で定義する．成分で書くと$(i,j)$成分は
    $$\Sigma_{ij}=\mathrm{Cov}(X_i,X_j)=E[(X_i-\mu_i)(X_j-\mu_j)]$$
    であり，特に対角成分$\Sigma_{ii}=\mathrm{Var}(X_i)$は各成分の分散である．
\end{definition}

\begin{remark}
    共分散行列は常に\textbf{半正定値対称行列}である．
    すなわち$\prescript{t}{}{}\Sigma=\Sigma$かつ任意の$\bm{v}\in\mathbb{R}^n$に対して
    $\prescript{t}{}{}\bm{v}\Sigma\bm{v}=\mathrm{Var}(\prescript{t}{}{}\bm{v}\bm{X})\geq 0$が成り立つ．
    $\bm{X}$の成分の間に完全な線形従属関係がないとき（非退化のとき），$\Sigma$は正定値対称行列になる．
\end{remark}

\begin{definition}[マハラノビス距離]
    $\bm{x},\bm{y}\in\mathbb{R}^n$と正定値対称行列$\Sigma\in M_n(\mathbb{R})$に対して，
    \textbf{マハラノビス距離}を
    $$d_M(\bm{x},\bm{y})=\sqrt{\prescript{t}{}{}(\bm{x}-\bm{y})\Sigma^{-1}(\bm{x}-\bm{y})}$$
    で定義する．
    特に確率分布$P$を考えるとき，$\Sigma$として$P$の共分散行列をとることが多い．
    $\Sigma=I$（単位行列）のときユークリッド距離に一致する．
\end{definition}

\begin{remark}
    マハラノビス距離はデータの\textbf{スケール}と\textbf{相関}を考慮した距離である．
    各成分の分散が異なる場合や成分間に相関がある場合に，
    ユークリッド距離より適切な距離を与える．
    \begin{enumerate}
        \item \textbf{スケール不変性}：
        各変数を標準化した場合でも同じ値を与える．
        すなわち$\bm{x}$の第$i$成分を$\sigma_i$倍しても距離は変わらない．
        \item \textbf{相関の補正}：
        $\Sigma^{-1}$によって変数間の相関を取り除いた空間でのユークリッド距離に相当する．
        より正確には，$\Sigma=A\,\prescript{t}{}{}A$と分解するとき（Cholesky分解など），
        $d_M(\bm{x},\bm{y})=\|A^{-1}(\bm{x}-\bm{y})\|$である．
        \item \textbf{距離の公理}：
        $\Sigma$が正定値対称行列ならば，$d_M$は距離の公理をすべて満たす．
    \end{enumerate}
\end{remark}




\section*{クォータニオン・回転行列・オイラー角}

\subsection*{クォータニオン}

\begin{definition}[クォータニオン（四元数）]
    虚数単位$i,j,k$が
    $$i^2=j^2=k^2=ijk=-1$$
    を満たすとき，
    $$\mathbb{H}=\{w+xi+yj+zk\mid w,x,y,z\in\mathbb{R}\}$$
    を\textbf{クォータニオン}（四元数）の集合という．
    $w$を\textbf{実部（スカラー部）}，$xi+yj+zk$を\textbf{虚部（ベクトル部）}という．
\end{definition}

\begin{definition}[共役・ノルム]
    $q=w+xi+yj+zk\in\mathbb{H}$に対して，
    \textbf{共役クォータニオン}を
    $$\bar{q}=w-xi-yj-zk$$
    で定義し，\textbf{ノルム}を
    $$\|q\|=\sqrt{q\bar{q}}=\sqrt{w^2+x^2+y^2+z^2}$$
    で定義する．$\|q\|=1$のとき$q$を\textbf{単位クォータニオン}という．
\end{definition}

\begin{remark}
    クォータニオンの積は結合的であるが一般に非可換である．
    $i,j,k$の積は次の表に従う：
    $$ij=k,\quad jk=i,\quad ki=j,\quad ji=-k,\quad kj=-i,\quad ik=-j.$$
    単位クォータニオン全体$S^3=\{q\in\mathbb{H}\mid\|q\|=1\}$は積について群をなす．
\end{remark}

\begin{definition}[クォータニオンによる回転]
    単位クォータニオン$q=\cos\dfrac{\theta}{2}+\sin\dfrac{\theta}{2}(n_x i+n_y j+n_z k)$
    （ただし$(n_x,n_y,n_z)$は単位ベクトル）に対して，
    $\mathbb{R}^3$のベクトル$\bm{v}=(v_x,v_y,v_z)$を
    純クォータニオン$p=v_x i+v_y j+v_z k$と同一視すると，
    単位ベクトル$\bm{n}=(n_x,n_y,n_z)$まわりの角度$\theta$の回転は
    $$p'=qp\bar{q}$$
    で与えられる．$p'$の虚部が回転後のベクトルに対応する．
\end{definition}

\begin{remark}
    $q$と$-q$は同一の回転を表す．すなわち単位クォータニオンと回転の対応は
    $2$対$1$であり，$\mathrm{Spin}(3)\cong S^3\to SO(3)$という
    二重被覆の構造をもつ．
\end{remark}

\subsection*{回転行列}

\begin{definition}[$SO(3)$（三次元回転群）]
    \textbf{三次元回転群}を
    $$SO(3)=\{R\in M_3(\mathbb{R})\mid R\,\prescript{t}{}{}R=I,\;\det R=1\}$$
    で定義する．各$R\in SO(3)$を\textbf{回転行列}という．
\end{definition}

\begin{definition}[基本回転行列]
    各座標軸まわりの角度$\theta$の回転行列を次で定義する：
    \begin{align*}
        R_x(\theta)
        &=\begin{pmatrix}1&0&0\\0&\cos\theta&-\sin\theta\\0&\sin\theta&\cos\theta\end{pmatrix},\\[6pt]
        R_y(\theta)
        &=\begin{pmatrix}\cos\theta&0&\sin\theta\\0&1&0\\-\sin\theta&0&\cos\theta\end{pmatrix},\\[6pt]
        R_z(\theta)
        &=\begin{pmatrix}\cos\theta&-\sin\theta&0\\\sin\theta&\cos\theta&0\\0&0&1\end{pmatrix}.
    \end{align*}
\end{definition}

\begin{proposition}[クォータニオンから回転行列への変換]
    単位クォータニオン$q=w+xi+yj+zk$（$w^2+x^2+y^2+z^2=1$）に対応する
    回転行列は
    $$R(q)=\begin{pmatrix}
        1-2(y^2+z^2) & 2(xy-wz) & 2(xz+wy) \\
        2(xy+wz) & 1-2(x^2+z^2) & 2(yz-wx) \\
        2(xz-wy) & 2(yz+wx) & 1-2(x^2+y^2)
    \end{pmatrix}$$
    である．
\end{proposition}

\subsection*{オイラー角}

\begin{definition}[オイラー角（ZYX 規約，航空機規約）]
    $\psi$（ヨー），$\theta$（ピッチ），$\phi$（ロール）を
    それぞれ$z$軸，$y$軸，$x$軸まわりの回転角とするとき，
    ZYX 規約の回転行列を
    $$R(\psi,\theta,\phi)=R_z(\psi)\,R_y(\theta)\,R_x(\phi)$$
    で定義する．成分を展開すると
    \begin{align*}
        R=\begin{pmatrix}
            c_\psi c_\theta
            & c_\psi s_\theta s_\phi - s_\psi c_\phi
            & c_\psi s_\theta c_\phi + s_\psi s_\phi \\
            s_\psi c_\theta
            & s_\psi s_\theta s_\phi + c_\psi c_\phi
            & s_\psi s_\theta c_\phi - c_\psi s_\phi \\
            -s_\theta
            & c_\theta s_\phi
            & c_\theta c_\phi
        \end{pmatrix}
    \end{align*}
    となる．ここで$c_\alpha=\cos\alpha$，$s_\alpha=\sin\alpha$と略記した．
\end{definition}

\begin{proposition}[回転行列からオイラー角への変換]
    $R=(r_{ij})$を ZYX 規約の回転行列とする（$\cos\theta\neq 0$，すなわち$\theta\neq\pm\pi/2$のとき）：
    \begin{align*}
        \psi &= \mathrm{atan2}(r_{21},\,r_{11}),\\
        \theta &= \mathrm{atan2}(-r_{31},\,\sqrt{r_{32}^2+r_{33}^2}),\\
        \phi &= \mathrm{atan2}(r_{32},\,r_{33}).
    \end{align*}
    $\cos\theta=0$のとき（ジンバルロック）は$\psi$と$\phi$が一意に定まらない．
\end{proposition}

\begin{remark}[ジンバルロック]
    $\theta=\pm\pi/2$のとき$R_y(\theta)$が$x$軸と$z$軸を重ね合わせるため，
    $\psi$と$\phi$の自由度が縮退して\textbf{ジンバルロック}が生じる．
    この状況では回転の一自由度が失われ，オイラー角による表現が不安定になる．
    クォータニオンによる表現ではジンバルロックが生じない．
\end{remark}

\begin{proposition}[オイラー角からクォータニオンへの変換]
    ZYX 規約のオイラー角$(\psi,\theta,\phi)$に対応する単位クォータニオンは
    \begin{align*}
        w &= \cos\tfrac{\phi}{2}\cos\tfrac{\theta}{2}\cos\tfrac{\psi}{2}
           +\sin\tfrac{\phi}{2}\sin\tfrac{\theta}{2}\sin\tfrac{\psi}{2},\\
        x &= \sin\tfrac{\phi}{2}\cos\tfrac{\theta}{2}\cos\tfrac{\psi}{2}
           -\cos\tfrac{\phi}{2}\sin\tfrac{\theta}{2}\sin\tfrac{\psi}{2},\\
        y &= \cos\tfrac{\phi}{2}\sin\tfrac{\theta}{2}\cos\tfrac{\psi}{2}
           +\sin\tfrac{\phi}{2}\cos\tfrac{\theta}{2}\sin\tfrac{\psi}{2},\\
        z &= \cos\tfrac{\phi}{2}\cos\tfrac{\theta}{2}\sin\tfrac{\psi}{2}
           -\sin\tfrac{\phi}{2}\sin\tfrac{\theta}{2}\cos\tfrac{\psi}{2}
    \end{align*}
    で与えられる．
\end{proposition}

\section*{確率}

\begin{definition}[確率空間・コルモゴロフの公理]
    集合$\Omega$（\textbf{標本空間}），$\sigma$-加法族$\mathcal{F}\subset 2^\Omega$（\textbf{事象族}），
    および写像$P\colon\mathcal{F}\to\mathbb{R}$（\textbf{確率測度}）の三つ組
    $(\Omega,\mathcal{F},P)$を\textbf{確率空間}という．
    $P$がコルモゴロフの公理を満たすとは，次の三条件が成り立つことをいう：
    \begin{enumerate}
        \item \textbf{非負性}：任意の$A\in\mathcal{F}$に対して$P(A)\geq 0$．
        \item \textbf{全確率}：$P(\Omega)=1$．
        \item \textbf{加算加法性}（$\sigma$-加法性）：
        互いに排反な事象の列$A_1,A_2,\ldots\in\mathcal{F}$
        （$i\neq j\Rightarrow A_i\cap A_j=\emptyset$）に対して
        $$P\!\left(\bigcup_{n=1}^{\infty}A_n\right)=\sum_{n=1}^{\infty}P(A_n).$$
    \end{enumerate}
\end{definition}

\begin{remark}
    コルモゴロフの公理から直ちに次が導かれる：
    \begin{enumerate}
        \item $P(\emptyset)=0$．
        \item \textbf{有限加法性}：$A\cap B=\emptyset$ならば$P(A\cup B)=P(A)+P(B)$．
        \item \textbf{余事象}：$P(A^c)=1-P(A)$．
        \item \textbf{単調性}：$A\subset B$ならば$P(A)\leq P(B)$．
        \item \textbf{劣加法性}（ブールの不等式）：
        $P\!\left(\bigcup_{n=1}^{\infty}A_n\right)\leq\sum_{n=1}^{\infty}P(A_n)$．
    \end{enumerate}
\end{remark}


\section*{微分方程式}

\subsection*{特異解と包絡線}

\begin{definition}[特異解]
    1階 ODE $F(x,y,y')=0$ の解で，
    積分定数 $c$ を含む一般解の族 $\Phi(x,y,c)=0$ のいずれにも属さない解を
    \textbf{特異解}という．
\end{definition}

\begin{definition}[包絡線]
    1パラメータ曲線族 $\Phi(x,y,c)=0$ に対し，
    族の各曲線に接する曲線を\textbf{包絡線}という．
    包絡線は\textbf{$c$-判別式法}，すなわち連立方程式
    \begin{equation*}
        \Phi(x,y,c)=0,\quad \frac{\partial\Phi}{\partial c}(x,y,c)=0
    \end{equation*}
    から $c$ を消去することで求められる（得られた曲線には尖点軌跡などを含む場合があるため，
    包絡線であるか否かは個別に確認が必要である）．
\end{definition}

\begin{proposition}[包絡線は特異解]\label{prop:envelope-singular}
    $F(x,y,p)=0$（$p=y'$）の一般解の族 $\Phi(x,y,c)=0$ の包絡線は，
    この ODE の特異解である．
\end{proposition}

\begin{proof}
    包絡線上の点 $(x_0,y_0)$ に対してある $c_0$ が存在し，
    \begin{equation*}
        \Phi(x_0,y_0,c_0)=0,\quad
        \frac{\partial\Phi}{\partial c}(x_0,y_0,c_0)=0.
    \end{equation*}
    $\Phi(x,y,c_0)=0$ は一般解の一員（解曲線）であるから，
    この曲線上の $(x_0,y_0)$ での傾きを $p_0$ とおくと $F(x_0,y_0,p_0)=0$．
    包絡線の定義より，包絡線は点 $(x_0,y_0)$ において $\Phi(x,y,c_0)=0$ と接する，
    すなわち同じ傾き $p_0$ をもつ．
    したがって $F(x_0,y_0,p_0)=0$ が包絡線上でも成り立ち，包絡線は ODE を満たす．
    包絡線は任意定数 $c$ を含まず一般解の族に属さないから，特異解である．
\end{proof}

\subsection*{クレロー方程式}

\begin{definition}[クレロー方程式]
    $p=y'$ とおくとき，
    \begin{equation*}
        y = xp + f(p)
    \end{equation*}
    の形の1階 ODE を\textbf{クレロー方程式}という．
\end{definition}

\begin{proposition}[クレロー方程式の包絡線は特異解]
    クレロー方程式 $y=xp+f(p)$ の一般解は直線族
    \begin{equation*}
        y = cx + f(c) \qquad (c\in\mathbb{R})
    \end{equation*}
    であり，この直線族の包絡線がクレロー方程式の特異解である．
    包絡線のパラメータ表示は
    \begin{equation*}
        x = -f'(c),\quad y = f(c) - cf'(c)
    \end{equation*}
    で与えられる．
\end{proposition}

\begin{proof}
    両辺を $x$ で微分すると
    \begin{equation*}
        p = p + \bigl(x + f'(p)\bigr)\frac{dp}{dx},
    \end{equation*}
    すなわち $\bigl(x+f'(p)\bigr)\,p'=0$．

    \textbf{場合1}（$p'=0$）：$p=c$（定数）とおくと $y=cx+f(c)$．
    これが積分定数 $c$ を含む一般解である．

    \textbf{場合2}（$x+f'(p)=0$）：$x=-f'(p)$ を元の方程式に代入すると
    $y=-pf'(p)+f(p)$．これは $p$ をパラメータとする特異解を与える．

    次に一般解の族 $\Phi(x,y,c)=y-cx-f(c)=0$ の包絡線を $c$-判別式法で求める：
    \begin{equation*}
        \frac{\partial\Phi}{\partial c} = -x - f'(c) = 0
        \implies x = -f'(c).
    \end{equation*}
    $\Phi=0$ に代入すると $y = cx + f(c) = -cf'(c) + f(c)$．
    これは場合2の特異解と完全に一致する．
    すなわち，クレロー方程式の特異解は一般解の包絡線であり，
    命題\uref{prop:envelope-singular}によってこれが特異解であることが保証される．
\end{proof}

\begin{example}
    $y = xp - p^2$（$f(p)=-p^2$）はクレロー方程式の例である．
    一般解は $y = cx - c^2$（放物線の接線群），
    特異解は $x = 2c$，$y = c^2$，すなわち $y = x^2/4$（放物線）であり，
    これは各接線の包絡線（放物線それ自身）に一致する．
\end{example}

\subsection*{ラグランジュ方程式}

\begin{definition}[ラグランジュ方程式]
    $p=y'$ とおくとき，
    \begin{equation*}
        y = x\phi(p) + \psi(p)
    \end{equation*}
    の形の1階 ODE を\textbf{ラグランジュ方程式}という（$\phi(p)\not\equiv p$）．
    $\phi(p)=p$ のときクレロー方程式に帰着する．
\end{definition}

\begin{proposition}[ラグランジュ方程式の包絡線は特異解]
    ラグランジュ方程式 $y=x\phi(p)+\psi(p)$ の両辺を $x$ で微分すると
    \begin{equation*}
        p - \phi(p) = \bigl[x\phi'(p)+\psi'(p)\bigr]\frac{dp}{dx}.
    \end{equation*}
    $p-\phi(p)\neq 0$ の領域では $p$ を独立変数とみて
    \begin{equation*}
        \frac{dx}{dp} - \frac{\phi'(p)}{p-\phi(p)}\,x = \frac{\psi'(p)}{p-\phi(p)}
    \end{equation*}
    という $x(p)$ に関する1階線形 ODE が得られる．
    その一般解 $x=X(p,c)$（積分定数 $c$ を含む）と
    $y=X(p,c)\,\phi(p)+\psi(p)$ が，$p$ をパラメータとして，ラグランジュ方程式の一般解を与える．
    一般解の族の包絡線（$c$-判別式法で求められる曲線）は，
    命題\uref{prop:envelope-singular}によって，ラグランジュ方程式の特異解となる．
\end{proposition}

\begin{remark}
    クレロー方程式とラグランジュ方程式に共通する本質的な構造は，
    「一般解が1パラメータ曲線族をなし，その包絡線が特異解を与える」という点にある．
    クレロー方程式では一般解が直線族となるため包絡線の計算が明示的に行えるが，
    ラグランジュ方程式では一般解がより複雑な曲線族となり，
    包絡線は $c$-判別式法を経て求める必要がある．
    いずれの場合も，包絡線が特異解であることの保証は命題\uref{prop:envelope-singular}による．
\end{remark}

\section{群の作用}
\begin{definition}
    群$G$が集合$X$に作用するとする．
    \begin{enumerate}
        \item $x\in X$のとき
        $\bm{Gx=\{gx\mid g\in G\}}$を
        $x$の$G$による\textbf{軌道}と呼ぶ．
        \item $x\in X$があり，
        $Gx=X$となるとき，この作用は
        \textbf{推移的}であるという．
        $X$は$G$の\textbf{等質空間}であるという．
        \item $x\in X$のとき
        $\bm{G_x=\{ g\in G \mid 
        gx=g\}}$
        と書き，$x$の\textbf{安定化群}と呼ぶ．
    \end{enumerate}
    \begin{proposition}
        $G$が集合$X$に作用し，
        $x,y\in X,g\in G$で
        $y=gx$なら，
        $Gy=Gx$かつ$G_y=gG_xg^{-1}$である．
    \end{proposition}
    簡単だから証明略．
    \begin{corollary}
        $G$が集合$X$に作用し，
        $x,y\in X$で
        $Gx\cap Gy\neq \emptyset$ならば
        $Gx=Gy,\, G_x\cong G_y$である．
    \end{corollary}
\end{definition}

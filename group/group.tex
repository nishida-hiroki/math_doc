\section{群の作用}
\begin{definition}
    群$G$が集合$X$に作用するとする．
    \begin{enumerate}
        \item $x\in X$のとき
        $\bm{Gx=\{gx\mid g\in G\}}$を
        $x$の$G$による\textbf{軌道}と呼ぶ．
        \item $x\in X$があり，
        $Gx=X$となるとき，この作用は
        \textbf{推移的}であるという．
        $X$は$G$の\textbf{等質空間}であるという．
        \item $x\in X$のとき
        $\bm{G_x=\{ g\in G \mid 
        gx=x\}}$
        と書き，$x$の\textbf{安定化群}と呼ぶ．
    \end{enumerate}
\end{definition}
\begin{proposition}
    $G$が集合$X$に作用し，
    $x,y\in X,g\in G$で
    $y=gx$なら，
    $Gy=Gx$かつ$G_y=gG_xg^{-1}$である．
\end{proposition}
    簡単だから証明略．
\begin{corollary}
    $G$が集合$X$に作用し，
    $x,y\in X$で
    $Gx\cap Gy\neq \emptyset$ならば
    $Gx=Gy,\, G_x\cong G_y$である．
\end{corollary}
\begin{corollary}
    群$G$が集合$X$に作用するとき，
    $x,y\in X$で$x\sim y \Leftrightarrow Gx=Gy$
    と定義する．
    すると$\sim$は$X$上の同値関係となる．
    この同値関係の剰余類は
    $X$上の軌道と$1$対$1$に対応する．
\end{corollary}
これも簡単
\begin{proposition}\label{1.5}
    $G$が集合$X$に作用するとする．$x\in X$であるとき，
    集合$Gx$と$G/G_x$は，対応
    $G/G_x\ni gG_x\mapsto gx\in Gx$
    により$1$対$1$に対応する．
    よって$|Gx|=(G\colon G_x)$である．
    有限群なら$|Gx|=|G|/|G_x|$である．
\end{proposition}
\begin{proof}
    まずwell-definedであることを示す．
    $g,g'\in G$に対して
    $gG_x=g'G_x$ならば，
    ある$h\in G_x$が存在して，
    $g=g'h$となる．
    このとき
    $gx=g'hx=g'x$よりwell-defined
    である．
    $x\in X$をひとつ固定する．
    単射性を示す．
    $gx=g'x$ならば
    $g'^{-1}gx=x$つまり
    $g'^{-1}g\in G_x$である．
    よって$gG_x=g'G_x$であり単射である．
    全射性は明らか．
\end{proof}

\begin{definition}
    $H$を群$G$の部分群とする．
    \begin{enumerate}
        \item $N_G(H)=\{g\in G\mid gHg^{-1}=H\}$
        \item $Z_G(H)=\{g\in G\mid \forall h \in H
        ,\, gh=hg\}$
        \item $Z(G)=Z_G(G)$
    \end{enumerate}
    と定義して$N_G(H),Z_G(H)$をそれぞれ
    $H$の\textbf{正規化群，中心化群}という．
    $Z(G)$は$G$の\textbf{中心}という．
\end{definition}

\begin{definition}
    群$G$の元$x,y$に対し，$g\in G$があり$y=gxg^{-1}$となるとき，
    $x$と$y$は\textbf{共役}であるという．
    $x$と共役である元の集合を$x$の\textbf{共役類}
    といい，$C(x)$とかく．
\end{definition}



\begin{theorem}\label{ruitousiki}
    $G$を有限群とする．
    \begin{enumerate}
        \item $x\in G$なら，$|C(x)|=
        |G|/|Z_G(x)|$である．
        また$C(x)=\{x\}$であることと
        $x$が$G$の中心$Z(G)$の元であること
        は同値である．
        \item 等式$|G|=\sum |C(x)|$が成り立つ．
        これを\textbf{類等式}という．
    \end{enumerate}
\end{theorem}



\begin{proof}
    (1)
    $g\in G,x\in G$に対して
    $g\circ x=gxg^{-1}$とすれば作用である．
    この軌道は$C(x)$であり，安定化群は
    $Z_G(x)$である．
    命題\uref{1.5}より成立．
    \begin{comment}
    $x$を固定する．
    $\psi\colon G\rightarrow C(x),\,g\mapsto g\circ x$
    を考えると軌道であるから明らかに全射である．    
    写像
    $$\bar{\psi}\colon G/Z_G(x)\rightarrow
    C(x),\,aZ_G(x)\mapsto \psi(a)$$
    を考える．
    well-defined性を示す．
    $aZ_G(x)=bZ_G(x)$のときある$h\in Z_G(x)$
    が存在して$a=bh$となる．このとき安定化群であることに注意
    して
    $\psi(a)=\psi(bh)=(bh)\circ x=b\circ(h\circ x)=b\circ
    x=\psi(b)\Leftrightarrow
    \bar{\psi}(aZ_G(x))=\bar{\psi}(bZ_G(x))$
    となる．よってwell-definedである．
    $\psi$が全射だったので
    $\bar{\psi}$も全射である．
    単射性を示す．
    $a,b\in G$に対して
    $\bar{\psi}(aZ_G(x))=\bar{\psi}(bZ_G(x))$
    が成り立つとする．このとき
    $\psi(a)=\psi(b)\Rightarrow a\circ x=b\circ
    x\Rightarrow b^{-1}a\circ x=x\Rightarrow 
    b^{-1}a\in Z_G(x)
    \Rightarrow aZ_G(x)=
    bZ_G(x)$が成り立つ．
    よって単射である．
    $\bar{\psi}$は全単射となり，
    両辺の個数を数えて(1)が示せた．\\
    \end{comment}
    (2)共役類は同値関係であるから明らか．
\end{proof}


類等式からつぎのことがわかる．
\begin{proposition}
    類等式の右辺の数字は次の性質をもつ．
    \begin{enumerate}
        \item 必ず$1$が$1$回は出現する．
        \item すべて$|G|$の約数である．
        \item $1$は$|G|$の約数個出現する．
    \end{enumerate}
    これらを\textbf{自明な考察}とよぶ．
\end{proposition}
\begin{proof}
    (1)$Z_G(e)=G$より$1$がすくなくともひとつある．\\
    (2)定理\uref{ruitousiki}を考えれば明らか．\\
    (3)共役な元が自分自身しかないことと中心の元であることは同値．
\end{proof}




\section{シロー}

\begin{proposition}
    $G$を有限群，$P(G)$を部分集合全体とする．さらに
    左作用を$G\times P(G)\rightarrow P(G);(g,S)\mapsto g\circ S=\{ gs\mid s\in S\}$
    と定める．このとき，
    部分集合$S\subset G$に対し
    $\lvert S\rvert \bigm| \lvert \operatorname{Stab}(S)\rvert$が成り立つ．
\end{proposition}

\begin{proof}
    $H=\operatorname{Stab}(S)$とおく．
    $s_i\in S$に対して$s_iH\subset S$であることより，
    $\displaystyle S=\bigcup_is_ie \subset \bigcup_i
    s_iH\subset S$が成り立つ．よって$\displaystyle \bigcup_i
    s_iH=S$である．$s_j\in S \setminus s_iH$をとると$s_jH\cap
    s_iH=\phi$である．これにより$\{ s_1,\ldots s_n\} \subset S$
    が存在して$\displaystyle\coprod_{i=1}^ns_iH=S$となる．
    よって$\lvert s_iH\rvert=\lvert H\rvert$であることから
    $\lvert S\rvert \bigm| \lvert H \rvert$が成り立つ．
\end{proof}
\begin{theorem}[シローの定理]
    $G$を有限群，$n=|G|$，$p$を$n$の素因数とし，
    $p^a$を$n$を割り切る最大の$p$冪とする．
    このとき次が成り立つ．
    \begin{enumerate}
        \item $G$の部分群$H$で$|H|=p^a$となるものが存在する．
        これを\textbf{シロー$p$部分群}とよぶ．
        \item シロー$p$部分群$H$をひとつ固定する．
        $K\subset G$が部分群で$|K|$が$p$冪なら
        ，$g\in G$があり，$K\subset gHg^{-1}$となる．
        特に$K$を含む$G$のシロー$p$部分群がある．
        \item $G$のすべてのシロー$p$部分群は共役である．
        \item シロー$p$部分群の数を$s$とすると
        $s=|G|/|N_G(H)|\equiv1\,(\mathrm{mod}\,p)$という
        条件を満たす．
    \end{enumerate}
\end{theorem}





\section{計算}
位数が素数$p$の群$G$に対して自明な考察を行えば各
$x_i\in G$に対して$|C(x_i)|=1$または$|C(x_i)|=p$であるが，
まず$|C(e)|=1$であることを考えれば，
任意の$x\in G$に対して$|C(x_i)|=1$である．
つまりアーベル群である．
単項で生成される部分群にラグランジュの定理を
使えば巡回群であることもわかる．
\\
また，位数が$p$冪の群$G$に対して自明な考察をする．
$|C(e)|=1$であり$|C(x)|$が$p$の倍数か$1$である．
これより$|C(x)|=1$となる$x$は$p$の倍数個ある．
つまり自明でない中心が存在することがわかる．\\
位数が$6$の群$G$に対して自明な考察をする．
候補は
\begin{enumerate}
    \item $6=1+1+1+1+1+1$
    \item $6=1+1+1+3$
    \item $6=1+1+2+2$
    \item $6=1+2+3$
\end{enumerate}
である．(1)が$\mathbb{Z}/6\mathbb{Z}$,(2)
については存在しない．理由について$|Z(G)|=3$であるから
$|G/Z(G)|=2$である．これは巡回群であるから，生成元を$gZ(G),g\in G $
とおいて考えると，
任意の$x,y\in G$にたいしてある$i,j\in \mathbb{N}$が
存在して$x=g^iz_1,y=g^jz_2$が成り立つ．ただし$z_1,z_2\in Z(G)$である．
$xy=g^iz_1g^jz_2=g^ig^jz_1z_2=g^jg^iz_2z_1=g^jz_2g^iz_1=yx$となり
可換群であるから$|Z(G)|=3$に矛盾する．(3)も(2)と同様に存在しない．
(4)は$S_3$である．

\section{群の作用}
\begin{definition}
    群$G$が集合$X$に作用するとする．
    \begin{enumerate}
        \item $x\in X$のとき
        $\bm{Gx=\{gx\mid g\in G\}}$を
        $x$の$G$による\textbf{軌道}と呼ぶ．
        \item $x\in X$があり，
        $Gx=X$となるとき，この作用は
        \textbf{推移的}であるという．
        $X$は$G$の\textbf{等質空間}であるという．
        \item $x\in X$のとき
        $\bm{G_x=\{ g\in G \mid 
        gx=x\}}$
        と書き，$x$の\textbf{安定化群}と呼ぶ．
    \end{enumerate}
\end{definition}
\begin{proposition}
    $G$が集合$X$に作用し，
    $x,y\in X,g\in G$で
    $y=gx$なら，
    $Gy=Gx$かつ$G_y=gG_xg^{-1}$である．
\end{proposition}
    簡単だから証明略．
\begin{corollary}
    $G$が集合$X$に作用し，
    $x,y\in X$で
    $Gx\cap Gy\neq \emptyset$ならば
    $Gx=Gy,\, G_x\cong G_y$である．
\end{corollary}
\begin{corollary}
    群$G$が集合$X$に作用するとき，
    $x,y\in X$で$x\sim y \Leftrightarrow Gx=Gy$
    と定義する．
    すると$\sim$は$X$上の同値関係となる．
    この同値関係の剰余類は
    $X$上の軌道と$1$対$1$に対応する．
\end{corollary}
これも簡単
\begin{proposition}
    $G$が集合$X$に作用するとする．$x\in X$であるとき，
    集合$Gx$と$G/G_x$は，対応
    $G/G_x\ni gG_x\mapsto gx\in Gx$
    により$1$対$1$に対応する．
    よって$|Gx|=(G\colon G_x)$である．
    有限群なら$|Gx|=|G|/|G_x|$である．
\end{proposition}
\begin{proof}
    まずwell-definedであることを示す．
    $g,g'\in G$に対して
    $gG_x=g'G_x$ならば，
    ある$h\in G_x$が存在して，
    $g=g'h$となる．
    このとき
    $gx=g'hx=g'x$よりwell-defined
    である．
    $x\in X$をひとつ固定する．
    単射性を示す．
    $gx=g'x$ならば
    $g'^{-1}gx=x$つまり
    $g'^{-1}g\in G_x$である．
    よって$gG_x=g'G_x$であり単射である．
    全射性は明らか．
\end{proof}

\begin{definition}
    $H$を群$G$の部分群とする．
    \begin{enumerate}
        \item $N_G(H)=\{g\in G\mid gHg^{-1}=H\}$
        \item $Z_G(H)=\{g\in G\mid \forall h \in H
        ,\, gh=hg\}$
        \item $Z(G)=Z_G(G)$
    \end{enumerate}
    と定義して$N_G(H),Z_G(H)$をそれぞれ
    $H$の\textbf{正規化群，中心化群}という．
    $Z(G)$は$G$の\textbf{中心}という．
\end{definition}

\begin{definition}
    群$G$の元$x,y$に対し，$g\in G$があり$y=gxg^{-1}$となるとき，
    $x$と$y$は\textbf{共役}であるという．
    $x$と共役である元の集合を$x$の\textbf{共役類}
    といい，$C(x)$とかく．
\end{definition}



\begin{theorem}\label{ruitousiki}
    $G$を有限群とする．
    \begin{enumerate}
        \item $x\in G$なら，$|C(x)|=
        |G|/|Z_G(x)|$である．
        また$C(x)=\{x\}$であることと
        $x$が$G$の中心$Z(G)$の元であること
        は同値である．
        \item 等式$|G|=\sum |C(x)|$が成り立つ．
        これを\textbf{類等式}という．
    \end{enumerate}
\end{theorem}

\begin{proof}
    (1)
    $g\in G,x\in G$に対して
    $g\circ x=gxg^{-1}$とすれば作用である．
    この軌道は$C(x)$であり，安定化群は
    $Z_G(x)$である．
    $x$を固定する．
    $\psi\colon G\rightarrow C(x),\,g\mapsto g\circ x$
    を考えると軌道であるから明らかに全射である．    
    写像
    $$\bar{\psi}\colon G/Z_G(x)\rightarrow
    C(x),\,aZ_G(x)\mapsto \psi(a)$$
    を考える．
    well-defined性を示す．
    $aZ_G(x)=bZ_G(x)$のときある$h\in Z_G(x)$
    が存在して$a=bh$となる．このとき安定化群であることに注意
    して
    $\psi(a)=\psi(bh)=(bh)\circ x=b\circ(h\circ x)=b\circ
    x=\psi(b)\Leftrightarrow
    \bar{\psi}(aZ_G(x))=\bar{\psi}(bZ_G(x))$
    となる．よってwell-definedである．
    $\psi$が全射だったので
    $\bar{\psi}$も全射である．
    単射性を示す．
    $a,b\in G$に対して
    $\bar{\psi}(aZ_G(x))=\bar{\psi}(bZ_G(x))$
    が成り立つとする．このとき
    $\psi(a)=\psi(b)\Rightarrow a\circ x=b\circ
    x\Rightarrow b^{-1}a\circ x=x\Rightarrow 
    b^{-1}a\in Z_G(x)
    \Rightarrow aZ_G(x)=
    bZ_G(x)$が成り立つ．
    よって単射である．
    $\bar{\psi}$は全単射となり，
    両辺の個数を数えて(1)が示せた．\\
    (2)共役類は同値関係であるから明らか．
\end{proof}


類等式からつぎのことがわかる．
\begin{proposition}
    類等式の右辺の数字は次の性質をもつ．
    \begin{enumerate}
        \item 必ず$1$が$1$は出現する．
        \item すべて$|G|$の約数である．
        \item $1$は$|G|$の約数個出現する．
    \end{enumerate}
\end{proposition}
\begin{proof}
    (1)$Z_G(e)=G$より$1$がすくなくともひとつある．\\
    (2),(3)定理\uref{ruitousiki}を考えれば明らか．
\end{proof}
